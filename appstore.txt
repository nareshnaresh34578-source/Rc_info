let apps = [];
let social = {};
let siteSettings = {};
let activeCategory = 'All';
let searchTerm = '';

function toast(msg){
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  setTimeout(()=>t.classList.remove('show'), 2200);
}

async function loadAll(){
  try{
    const a = await window.storage.get('apps', true);
    apps = a ? JSON.parse(a.value) : [];
  }catch(e){ apps = []; }
  try{
    const s = await window.storage.get('social-links', true);
    social = s ? JSON.parse(s.value) : {};
  }catch(e){ social = {}; }
  try{
    const st = await window.storage.get('site-settings', true);
    siteSettings = st ? JSON.parse(st.value) : {};
  }catch(e){ siteSettings = {}; }
  applySiteSettings();
  renderSocial();
  renderChips();
  renderGrid();
  await checkNewApps();
}

async function getSeenIds(){
  try{
    const r = await window.storage.get('seen-app-ids', false);
    return r ? JSON.parse(r.value) : [];
  }catch(e){ return []; }
}

async function setSeenIds(ids){
  try{
    await window.storage.set('seen-app-ids', JSON.stringify(ids), false);
  }catch(e){ /* ignore */ }
}

async function checkNewApps(){
  const seen = await getSeenIds();
  const seenSet = new Set(seen.map(String));
  const newOnes = apps.filter(a => !seenSet.has(String(a.id)));
  const banner = document.getElementById('notifyBanner');
  const text = document.getElementById('notifyText');

  // First-ever visit: nothing to announce, just mark everything seen.
  if(seen.length === 0 && apps.length > 0){
    await setSeenIds(apps.map(a=>a.id));
    return;
  }
  if(newOnes.length === 0){
    banner.classList.remove('show');
    return;
  }
  const names = newOnes.map(a=>a.name);
  let msg;
  if(names.length === 1){
    msg = `<b>New app available:</b> ${escapeHtml(names[0])}`;
  }else if(names.length <= 3){
    msg = `<b>New apps available:</b> ${names.map(escapeHtml).join(', ')}`;
  }else{
    msg = `<b>${names.length} new apps available</b> since your last visit`;
  }
  text.innerHTML = msg;
  banner.classList.add('show');
}

document.getElementById('notifyDismiss').onclick = async ()=>{
  document.getElementById('notifyBanner').classList.remove('show');
  await setSeenIds(apps.map(a=>a.id));
};

function applySiteSettings(){
  document.getElementById('siteName').textContent = siteSettings.name || 'App Workshop';
  document.getElementById('siteTag').textContent = siteSettings.tag || 'apps built & shipped by one person';
  document.getElementById('heroHeadline').textContent = siteSettings.headline || 'A small workshop for the things I build.';
  document.getElementById('heroSub').textContent = siteSettings.intro || "Everything here is made, maintained, and shipped by me. Browse the catalog below, grab what's useful, and find me elsewhere through the links up top.";
  document.title = (siteSettings.name || 'App Workshop');
}

function renderSocial(){
  const row = document.getElementById('socialRow');
  row.innerHTML = '';
  const map = [
    ['whatsapp', social.whatsapp], ['telegram', social.telegram],
    ['github', social.github], ['twitter', social.twitter], ['instagram', social.instagram],
    ['linkedin', social.linkedin], ['youtube', social.youtube],
    ['email', social.email ? 'mailto:'+social.email : ''], ['website', social.website]
  ];
  map.forEach(([key, url])=>{
    if(!url) return;
    const a = document.createElement('a');
    a.href = url; a.target = '_blank'; a.rel = 'noopener noreferrer';
    a.className = 'social-btn'; a.title = key;
    a.innerHTML = ICONS[key];
    row.appendChild(a);
  });
}

function renderChips(){
  const cats = ['All', ...new Set(apps.map(a=>a.category).filter(Boolean))];
  const row = document.getElementById('chipsRow');
  row.innerHTML = '';
  cats.forEach(c=>{
    const b = document.createElement('button');
    b.className = 'chip' + (c===activeCategory ? ' active' : '');
    b.textContent = c;
    b.onclick = ()=>{ activeCategory = c; renderChips(); renderGrid(); };
    row.appendChild(b);
  });
  document.getElementById('statCats').textContent = cats.length - 1;
}

function isImageUrl(v){
  return /^https?:\/\//i.test(v || '');
}

function renderGrid(){
  const grid = document.getElementById('grid');
  const empty = document.getElementById('emptyState');
  let filtered = apps.filter(a=>{
    const matchCat = activeCategory === 'All' || a.category === activeCategory;
    const matchSearch = !searchTerm || (a.name + a.tagline + a.category).toLowerCase().includes(searchTerm.toLowerCase());
    return matchCat && matchSearch;
  });
  document.getElementById('statCount').textContent = apps.length;
  grid.innerHTML = '';
  if(filtered.length === 0){
    empty.style.display = 'block';
    return;
  }
  empty.style.display = 'none';
  filtered.slice().reverse().forEach(app=>{
    const card = document.createElement('div');
    card.className = 'card';
    const iconHtml = isImageUrl(app.icon)
      ? `<img src="${app.icon}" alt="">`
      : (app.icon || '📦');
    card.innerHTML = `
      ${app.version ? `<div class="stamp mono">v${escapeHtml(app.version)}</div>` : ''}
      <div class="card-top">
        <div class="app-icon">${iconHtml}</div>
        <div>
          <div class="app-name">${escapeHtml(app.name)}</div>
          <div class="app-cat">${escapeHtml(app.category || 'App')}${app.platform ? ' · ' + escapeHtml(app.platform) : ''}</div>
        </div>
      </div>
      <div class="app-tagline">${escapeHtml(app.tagline || '')}</div>
      <div class="card-footer">
        <a class="download-btn" href="${app.url || '#'}" target="_blank" rel="noopener noreferrer">Get app</a>
        <button class="del-btn" title="Remove" data-id="${app.id}">
          <svg width="15" height="15" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M3 6h18M8 6V4a1 1 0 0 1 1-1h6a1 1 0 0 1 1 1v2m2 0v14a2 2 0 0 1-2 2H7a2 2 0 0 1-2-2V6h14z"/></svg>
        </button>
      </div>
    `;
    grid.appendChild(card);
  });
  grid.querySelectorAll('.del-btn').forEach(btn=>{
    btn.onclick = ()=> removeApp(btn.dataset.id);
  });
}

function escapeHtml(str){
  const d = document.createElement('div');
  d.textContent = str || '';
  return d.innerHTML;
}

async function removeApp(id){
  apps = apps.filter(a=>String(a.id) !== String(id));
  try{
    await window.storage.set('apps', JSON.stringify(apps), true);
    renderChips(); renderGrid();
    toast('App removed');
  }catch(e){ toast('Could not save — try again'); }
}

  const name = document.getElementById('f_name').value.trim();
  const url = document.getElementById('f_url').value.trim();
  if(!name){ toast('Give the app a name first'); return; }
  const newApp = {
    id: Date.now(),
    name,
    icon: document.getElementById('f_icon').value.trim(),
    category: document.getElementById('f_category').value.trim() || 'App',
    tagline: document.getElementById('f_tagline').value.trim(),
    version: document.getElementById('f_version').value.trim(),
    platform: document.getElementById('f_platform').value.trim(),
    url: url || '#'
  };
  apps.push(newApp);
  try{
    await window.storage.set('apps', JSON.stringify(apps), true);
    const seen = await getSeenIds();
    await setSeenIds([...seen, newApp.id]);
    appOverlay.classList.remove('show');
    ['f_name','f_icon','f_category','f_tagline','f_version','f_platform','f_url'].forEach(id=>document.getElementById(id).value='');
    renderChips(); renderGrid();
    toast(`"${newApp.name}" is live in the catalog`);
  }catch(e){ toast('Could not save — try again'); }
};
